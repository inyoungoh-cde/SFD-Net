#     uv venv --python 3.11
#     uv pip install -r requirements.txt
#
# torch is pinned by wheel URL rather than through --extra-index-url. The
# PyTorch index also mirrors numpy, tqdm and setuptools, but at older versions

#
# For Windows, or a different Python version, substitute the matching wheel
# from https://download.pytorch.org/whl/cu124/torch/ (cp311 = Python 3.11).
#
# The CUDA extension `pointnet2_ops` is not listed here: it compiles against
# the torch this file installs, so it has to be a second step. See the
# "CUDA extension" section of README.md.

torch @ https://download.pytorch.org/whl/cu124/torch-2.4.1%2Bcu124-cp311-cp311-linux_x86_64.whl
numpy==2.4.6
tqdm==4.69.1

# Required in the environment when building pointnet2_ops with
# --no-build-isolation (step 2 in the README).
